\begin{block}{The Problem}
The KV cache needs $\mathcal{O}(2LHdT)$ floats and is \textbf{the} memory
bottleneck at 128K+ context. Low-rank compression stores
$K \approx U\Sigma V^\top$ at rank $R \ll d$, but online methods face a tradeoff:
\vspace{0.3cm}
\begin{itemize}
  \item \textbf{Oja's rule (OjaKV)}: cheap, but a stochastic-gradient step
        --- no per-token optimality, basis \textbf{drifts} at long context.
  \item \textbf{Brand's rank-1 SVD update}: exact and EYM-optimal, but costs
        $\mathcal{O}(R^3 + TR^2)$ per token --- \textbf{grows with $T$}.
\end{itemize}
\takeaway{\textbf{Idea:} be exact only while it is cheap, then \textbf{freeze}.}
\end{block}

\begin{block}{Method: Warmup-Then-Freeze}
\begin{center}
\begin{tikzpicture}[node distance=0.9cm]
  \node[box=teal]  (phi)   {\textbf{$\Phi$-bridge}: one Gaussian $\Phi\in\R^{R\times d}$ is both\\
                            importance sketch \emph{and} Halko range finder};
  \node[box=navy,below=of phi]  (cold)  {\textbf{Halko cold start} on prefix\\
                            $Y{=}\mathrm{diag}(w)K\Phi^\top$, QR, small SVD};
  \node[box=accent,below=of cold] (warm) {\textbf{Warmup} ($t \le T_0$): exact Brand rank-1 update\\
                            EYM-optimal after \emph{every} token};
  \node[box=good,below=of warm] (frz)  {\textbf{Freeze} ($t > T_0$): $\mathbf{c} = V_K\mathbf{k}$, append\\
                            $\mathcal{O}(Rd)$ per token --- no SVD, no rotation};
  \draw[arr] (phi) -- (cold);
  \draw[arr] (cold) -- (warm);
  \draw[arr] (warm) -- (frz);
\end{tikzpicture}
\end{center}
\vspace{0.3cm}
Attention is computed straight from the factors,
$\mathbf{s} = \mathbf{q} V_K^\top \Sigma_K U_K^\top$, at
$\mathcal{O}(R(T{+}d))$ instead of $\mathcal{O}(Td)$ --- identically before
and after the freeze.
\end{block}

\begin{block}{The $\Phi$-Bridge}
Token-importance sketching uses a random Gaussian $\Phi$ with
$\Phi\mathbf{x}\in\R^R$. Halko's randomized range finder needs \emph{exactly}
the same object: a random $\Omega\in\R^{d\times R}$ with $K\Omega$ spanning $K$'s range.
\takeaway{Set $\Omega = \Phi^\top$: one matrix, two roles,\\ no separate SVD warm-up cost.}
Valid because Gaussian $\Phi$ satisfies both Johnson--Lindenstrauss
concentration and Halko's sub-Gaussian-row requirement.
\end{block}

\begin{block}{Guarantees}
\textbf{Warmup ($t\le T_0$): per-token EYM optimality.}
\[
\| K_t - U_t\,\mathrm{diag}(\boldsymbol{\sigma}_t)\,V_t^\top \|_F
  = \min_{\mathrm{rank}(B)\le R} \| K_t - B \|_F
\]
\textbf{Post-freeze ($t > T_0$):} $\mathcal{O}(Rd)$ per token, exact within
$\mathrm{rowspace}(V_{T_0})$; no fresh optimality guarantee.
OjaKV has neither.

\vspace{0.3cm}
\textbf{XLA-static shapes:} a fixed $T_{\max}{\times}R$ buffer written via
\texttt{dynamic\_update\_slice} gives the first GPU/TPU-compilable Brand update.
\end{block}

\end{column}

\begin{block}{Exact Updates Beat Oja's Rule}

\end{block}

\begin{block}{Limitations \& Next Steps}
\begin{itemize}
  \item Long-context runs use synthetic, non-shifting statistics.
  \item Perplexity on a small held-out set; WikiText-2 / PG-19 pending.
  \item CPU timings only; GPU, Llama, Qwen needed.
\end{itemize}
\end{block}

\begin{block}{Take-Home}
\takeaway{Exact updates while cheap, a frozen basis after:\\
\textbf{TKV is near-lossless where Oja's rule collapses,}\\
at 8.5$\times$ less KV memory.}
\end{block}
